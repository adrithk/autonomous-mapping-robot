cmake_minimum_required(VERSION 3.8)
project(my_bot)
# Optimize the Pi-side plugin/transport while retaining debugging symbols.
# Respect explicitly selected builds and multi-configuration generators.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(Threads REQUIRED)
add_library(esp32_transport STATIC src/serial_transport.cpp)
set_target_properties(esp32_transport PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(esp32_transport PUBLIC cxx_std_17)
target_link_libraries(esp32_transport PUBLIC Threads::Threads)
target_include_directories(esp32_transport PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
add_library(esp32_hardware SHARED src/esp32_system.cpp)
target_link_libraries(esp32_hardware esp32_transport)
ament_target_dependencies(esp32_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface esp32_hardware.xml)
install(TARGETS esp32_hardware esp32_transport
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(FILES HARDWARE.md SIMULATION.md WSL_SETUP.md SLAM.md NAVIGATION.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config description launch worlds DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/teleop_wasd.py DESTINATION lib/${PROJECT_NAME} RENAME teleop_wasd)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(model_checks test/test_model.py)
  ament_add_pytest_test(rviz_launch_checks test/test_rviz_launch.py)
  ament_add_pytest_test(navigation_checks test/test_navigation.py)
  ament_add_pytest_test(teleop_checks test/test_teleop.py)
  add_executable(hardware_plugin_test test/hardware_plugin_test.cpp)
  target_compile_features(hardware_plugin_test PRIVATE cxx_std_17)
  ament_target_dependencies(hardware_plugin_test hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  target_compile_options(hardware_plugin_test PRIVATE -UNDEBUG)
  add_test(NAME hardware_plugin_test COMMAND hardware_plugin_test)
  add_executable(serial_transport_test test/serial_transport_test.cpp)
  target_link_libraries(serial_transport_test esp32_transport Threads::Threads util)
  target_compile_options(serial_transport_test PRIVATE -UNDEBUG)
  add_test(NAME serial_transport_test COMMAND serial_transport_test)
endif()
ament_package()
